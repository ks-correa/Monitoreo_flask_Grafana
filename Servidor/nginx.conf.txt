
# Grupo de instancias Flask.
# Nginx distribuye las solicitudes mediante Round Robin.
upstream servidores_flask {
    zone servidores_flask 64k;
    # DNS interno de Docker: actualizar IPs sin reiniciar NGINX.
    resolver 127.0.0.11 valid=5s ipv6=off;
    server flask1:5000 resolve;
    server flask2:5000 resolve;
    server flask3:5000 resolve;
}

server {
    # Puerto interno de Nginx.
    listen 80;
    server_name _;

    location / {
        # Enviar las solicitudes al grupo Flask.
        proxy_pass http://servidores_flask;
        # Reintentar pronto en otra réplica si la IP detenida no responde.
        proxy_connect_timeout 1s;

        # Conservar información de la petición original.
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
