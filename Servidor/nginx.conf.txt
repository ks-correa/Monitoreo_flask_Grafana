
# Grupo de instancias Flask.
# Nginx distribuye las solicitudes mediante Round Robin.
upstream servidores_flask {
    server flask1:5000;
    server flask2:5000;
    server flask3:5000;
}

server {
    # Puerto interno de Nginx.
    listen 80;
    server_name _;

    location / {
        # Enviar las solicitudes al grupo Flask.
        proxy_pass http://servidores_flask;

        # Conservar información de la petición original.
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
